% ---------- Outline ----------
% Outline after the title, then again at each later section with the
% current section in the accent colour and the rest in grey. Outline
% slides are unnumbered and don't advance the slide count.
\setbeamertemplate{section in toc}{%
  {\color{muted}\inserttocsectionnumber.}\hspace{0.6em}\inserttocsection\par}
\setbeamercolor{section in toc}{fg=ink}
\setbeamertemplate{section in toc shaded}[default][100]
\setbeamercolor{section in toc shaded}{fg=muted}
\AtBeginSection{%
  \ifnum\value{section}>1
    {\setbeamercolor{section in toc}{fg=accent}
     \setbeamertemplate{footline}{}
     \begin{frame}[noframenumbering]{Outline}
       \tableofcontents[currentsection]
     \end{frame}}
  \fi}

% Cross-references to slides by their visible number, e.g. "slide 14".
% Put \slidelabel{name} inside a frame; \slideref{name} prints its number.
\makeatletter
\newcommand{\slidelabel}[1]{%
  \ifnum\beamer@slideinframe=1
    \protected@write\@auxout{}{\string\newlabel{slide:#1}{{\insertframenumber}{\thepage}}}%
  \fi}
\makeatother
\newcommand{\slideref}[1]{\ref{slide:#1}}
