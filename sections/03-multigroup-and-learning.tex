\section{Multigroup fairness and learning}

\begin{frame}{Relation to learning: changing perspectives}
  We've viewed groups $\G$ as subgroups of the population
  \textcolor{muted}{(potentially overlapping).}

  Equally, think of them as a class of functions
  $\C$, \ $c : \X \to \{0,1\}$:\\
  a hypothesis class, as in agnostic learning
  \textcolor{muted}{(slide~\slideref{relative-guarantee}; halfspaces, decision trees, \dots)}.
  \slidecite{Trevisan, Tulsiani, Vadhan '09}

  \pause
  \vspace{0.8em}
  Suppose $p$ is \alert{multiaccurate} for $\C$:
  \[
    \big|\,\E\big[\,c(x)\,(y - p(x))\,\big]\,\big| \approx 0 \qquad \forall c \in \C .
  \]

  An \alert{indistinguishability} condition: no $c \in \C$ correlates
  with the residual $y - p(x)$.
\end{frame}

\begin{frame}{Multicalibration as agnostic learning}\slidelabel{mc-agnostic}
  Suppose $p : \X \to [0,1]$ is \alert{multicalibrated} for $\C$:
  \[
    \big|\,\E\big[\,c(x)\,(y - p(x)) \mid p(x) = v\,\big]\,\big| \approx 0
    \qquad \forall c \in \C,\ \forall v .
  \]

  \begin{columns}[c]
    \begin{column}{0.45\textwidth}
      \centering
      \begin{tikzpicture}
        \begin{scope}
          \clip (0,0) ellipse (2.3 and 1.3);
          \fill[muted!18] (0.15,-1.3) rectangle (0.85,1.3);
        \end{scope}
        \draw[space] (0,0) ellipse (2.3 and 1.3);
        \node[lab] at (2.2,1.3) {$\X$};
        \foreach \x in {-1.6,-0.6,0.15,0.85,1.7} {
          \pgfmathsetmacro{\h}{1.3*sqrt(1-(\x/2.3)^2)}
          \draw[strip] (\x,-\h) -- (\x,\h);
        }
        \node[note] at (0.5,-1.85) {$p(x) = v$\\(constant on every strip)};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.53\textwidth}
      On every strip,
      \[
        \big|\,\E\big[\,c(x)\,(y - v)\,\big]\,\big| \approx 0 .
      \]

      To functions in $\C$, on a strip, $(x,y)$ looks like $(x,\tilde y)$
      where $\tilde y \sim \Ber(v)$.

      \vspace{0.4em}
      This is \alert{swap agnostic learning}: stronger than agnostic learning
      \textcolor{muted}{(slide~\slideref{swap})}.
    \end{column}
  \end{columns}

  \vspace{1.2em}
  \alert{A computational analogue of learning up to irreducible noise.}
  \textcolor{muted}{(slide~\slideref{predictor-view})}
\end{frame}

\begin{frame}{How to construct such predictors}
  Start with any predictor $p$.

  \textcolor{muted}{For fairness applications, start with a $p$ that has
  high accuracy / low loss.}

  \vspace{0.8em}
  Make $p$ multiaccurate / multicalibrated \alert{without increasing}
  \[
    \E\big[\,(p(x) - y)^2\,\big] .
  \]

  \vspace{0.5em}
  \alert{Key step:} find a $c \in \C$ ($g \in \G$) correlated with
  $y - p(x)$, a learning problem.

  \slidecite{H\'ebert-Johnson, Kim, Reingold, Rothblum '17}
\end{frame}

\begin{frame}{Weak agnostic learning}\slidelabel{weak-learning}
  $\D$ a distribution over $\X \times \{0,1\}$; \ $\C$ a class of functions.

  \vspace{0.6em}
  \alert{$(\alpha,\beta)$-weak agnostic learner}, for $\alpha \ge \beta > 0$:
  \begin{itemize}
    \item if there is a $c \in \C$ with $\err(c) \le \tfrac12 - \alpha$,
    \item then output $h : \X \to \{0,1\}$ with $\err(h) \le \tfrac12 - \beta$.
  \end{itemize}
  \textcolor{muted}{Equivalently: $c$ has correlation $\ge 2\alpha$ with the labels.
  Weak learners find correlations.}

  \pause
  \vspace{0.6em}
  \textbf{Theorem} \textcolor{muted}{(Kalai--\textcolor{ink}{K.}\ '09; Feldman '10)}\\
  Weak agnostic learning is equivalent to strong agnostic learning,
  \alert{even for a fixed marginal distribution over $\X$}.
\end{frame}

\begin{frame}{Auditing for multiaccuracy violations}
  If $p : \X \to [0,1]$ is not multiaccurate, there is a $c \in \C$ with
  \[
    \big|\,\E\big[\,c(x)\,(y - p(x))\,\big]\,\big| \ \text{noticeable.}
  \]

  \alert{Relabel}: keep $x$, replace $y$ by $\tilde y \in \{0,1\}$ with
  \[
    \Prob[\tilde y = 1 \mid x, y] = \tfrac12\big(1 + y - p(x)\big),
    \quad \text{so} \quad
    \E[\,2\tilde y - 1 \mid x, y\,] = y - p(x)
  \]
  and hence
  \[
    \E\big[\,c(x)\,(y - p(x))\,\big] = \E\big[\,c(x)\,(2\tilde y - 1)\,\big] .
  \]
  Run a weak agnostic learner on $(x, \tilde y)$ to find $h$.

  The \alert{marginal over $\X$ is unchanged}: only the labels change
  \textcolor{muted}{(slide~\slideref{weak-learning})}.

  \slidecite{H\'ebert-Johnson, Kim, Reingold, Rothblum '17}
\end{frame}

\begin{frame}{Correcting the violation}
  Update:
  \[
    p \;\leftarrow\; \Pi_{[0,1]}\big(p + \eta\, h\big)
  \]
  \textcolor{muted}{$\Pi_{[0,1]}(f)$ clips the range of $f$ to $[0,1]$.}

  \vspace{0.5em}
  Can show that the squared loss $\E\big[\,(p(x) - y)^2\,\big]$ drops noticeably.

  \vspace{0.8em}
  Repeat until the auditor finds nothing. The loss is $\ge 0$,
  so \alert{this stops}.

  \slidecite{H\'ebert-Johnson, Kim, Reingold, Rothblum '17}
\end{frame}

\begin{frame}{Multigroup fairness \& learning}\slidelabel{swap}
  \begin{columns}[c]
    \begin{column}{0.58\textwidth}
      Multicalibration with $\C$ gives a strong form of agnostic learning:
      \alert{swap agnostic learning}
      \textcolor{muted}{(Gopalan--Kim--Reingold '23)}.

      \vspace{0.4em}
      $p$ competes with the best $c \in \C$ \alert{separately on each
      level set} of $p$, not just one $c$ globally.
      \textcolor{muted}{(slide~\slideref{mc-agnostic})}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.8]
        \draw[space] (0,0) ellipse (2.3 and 1.3);
        \node[lab] at (2.2,1.3) {$\X$};
        \foreach \x in {-1.6,-0.6,0.15,0.85,1.7} {
          \pgfmathsetmacro{\h}{1.3*sqrt(1-(\x/2.3)^2)}
          \draw[strip] (\x,-\h) -- (\x,\h);
        }
        \foreach \x/\i in {-1.1/1,-0.225/2,0.5/3,1.275/4} {
          \node[font=\scriptsize, text=accent] at (\x,0) {$c_{\i}$};
        }
      \end{tikzpicture}
    \end{column}
  \end{columns}

  \vspace{0.6em}
  But \alert{costly}: auditing every level set takes many calls
  to the weak learner.

  \pause
  \vspace{1.5em}
  \alert{What about just multiaccuracy?}

  \alert{Much cheaper} to achieve: avoids having to audit level set by level set.
\end{frame}
