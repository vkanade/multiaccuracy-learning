\begin{frame}{Agnostic learning: a relative guarantee}\slidelabel{relative-guarantee}
  \begin{columns}[c]
    \begin{column}{0.56\textwidth}
      Data $\big((x_1,y_1),\dots,(x_n,y_n)\big) \sim \D^n$

      \vspace{0.3em}
      Comparator class $\F$ of functions $f : \X \to \{0,1\}$

      \vspace{1em}
      With probability at least $1-\delta$,
      \[
        \err(h) \;\le\;
        \underbrace{\min_{f \in \F} \err(f)}_{\textcolor{muted}{\text{best in class}}}
        \;+\; \varepsilon .
      \]
    \end{column}

\begin{frame}{Agnostic learning: predicting the truth}
  Learn a predictor $p : \X \to [0,1]$.

  \vspace{0.3em}
  The holy grail:
  \[
    p(x) = p^*(x) = \E[\,y \mid x\,]
  \]

  \pause
  \vspace{0.8em}
  Two sources of uncertainty:
  \begin{itemize}
    \item \alert{Epistemic}: $p \neq p^*$, since $p^*$ may be too complex
      and we see only finite data.
    \item \alert{Aleatoric}: $p^* \neq y$, since $y$ is inherently random
      given $x$.
  \end{itemize}
\end{frame}

\begin{frame}{Agnostic learning: the predictor view}\slidelabel{predictor-view}
  Holy grail: given $x$, $y$ is just a coin flip.
  \[
    y \mid \alert{x} \;\mathrel{\textcolor{accent}{\sim}}\; \Ber\big(p^*(x)\big)
  \]

  \pause
  \vspace{0.5em}
  \alert{Computational analogue}: we can't get $p^*$, but we can get some $p$ with
  \[
    y \mid \alert{p(x)} \;\mathrel{\textcolor{accent}{\approx_{\C}}}\; \Ber\big(p(x)\big)
  \]
  {\color{muted}$\approx_{\C}$: indistinguishable to every function in $\C$.}

  \vspace{1.5em}
  \textcolor{muted}{We'll return to this on slide~\slideref{mc-agnostic}, after a detour through multigroup fairness.}
\end{frame}
