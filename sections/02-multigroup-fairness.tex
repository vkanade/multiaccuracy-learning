\begin{frame}{Multigroup fairness notions: accuracy in expectation}
  A prediction problem: $(x,y) \sim \D$, \ $y \in \{0,1\}$
  \textcolor{muted}{\quad e.g.\ the outcome of an exam}

  \vspace{0.5em}
  A predictor $p : \X \to [0,1]$ gives the probability of a positive outcome.

  \pause
  \vspace{0.8em}
  $p$ is \alert{accurate in expectation} if
  \[
    \E[\,p(x)\,] = \E[\,y\,].
  \]

  \pause
  Expected accuracy error:
  \[
    \EAE(p) = \big|\,\E[\,p(x) - y\,]\,\big|
  \]
\end{frame}

\begin{frame}{Multigroup fairness notions: calibration}
  $p$ is \alert{calibrated} if
  \[
    \E[\,y \mid p(x)\,] = p(x).
  \]

  \pause
  Expected calibration error:
  \[
    \ECE(p) = \E\Big[\,\big|\,\E[\,y \mid p(x)\,] - p(x)\,\big|\,\Big]
  \]
\end{frame}
