\section{Multigroup fairness}

\begin{frame}{Accuracy in expectation}
  A prediction problem: $(x,y) \sim \D$, \ $y \in \{0,1\}$
  \textcolor{muted}{\quad e.g.\ the outcome of an exam}

  \vspace{0.5em}
  $p(x)$: predicted probability of a positive outcome.

  \pause
  \vspace{0.8em}
  $p$ is \alert{accurate in expectation} if
  \[
    \E[\,p(x)\,] = \E[\,y\,].
  \]

  Expected accuracy error:
  \[
    \EAE(p) = \big|\,\E[\,p(x) - y\,]\,\big|
  \]

  \textcolor{muted}{A global notion, and a very weak requirement:
  the constant $p \equiv \E[y]$ satisfies it.}
\end{frame}

\begin{frame}{Calibration}\slidelabel{calibration}
  $p$ is \alert{calibrated} if
  \[
    \E[\,y \mid p(x)\,] = p(x).
  \]

  Expected calibration error:
  \[
    \ECE(p) = \E\Big[\,\big|\,\E[\,y \mid p(x)\,] - p(x)\,\big|\,\Big]
  \]

  \textcolor{muted}{Also a global notion: the constant $p \equiv \E[y]$
  is calibrated too.}
\end{frame}

  \]

\begin{frame}{Multiaccuracy: the definition}
  Groups $\G = \{g_1, \dots, g_k\}$, possibly overlapping.

  \vspace{0.3em}
  Treat each $g$ also as its indicator function $g : \X \to \{0,1\}$.

  \vspace{0.8em}
  $p$ is \alert{$\tau$-multiaccurate} if
  \[
    \big|\,\E[\,g(x)\,(p(x) - y)\,]\,\big| \le \tau
    \qquad \forall g \in \G .
  \]
  \vspace{-0.7em}
  {\color{muted}
  \[
    \E[\,g(x)\,(p(x) - y)\,] = \Prob[x \in g] \cdot \E[\,p(x) - y \mid x \in g\,]
  \]
  Small groups get proportionally more slack: no conditioning on very small groups.}

  \pause
  \vspace{0.6em}
  \alert{Fairness:} no group is systematically over- or under-estimated
  on average.
\end{frame}

\begin{frame}{Multicalibration: the definition}
  For any group $g$ and any $v \in [0,1]$,
  \[
    \big|\,\E[\,p(x) - y \mid p(x) = v,\ x \in g\,]\,\big| \approx 0 .
  \]
  Calibration must hold even conditioned on group membership.

  \vspace{0.5em}
  As with multiaccuracy, avoid conditioning on small groups.

  $p$ is \alert{$\tau$-multicalibrated} if
  \[
    \big|\,\E[\,g(x)\,(p(x) - y) \mid p(x) = v\,]\,\big| \le \tau
    \qquad \forall g \in \G,\ \forall v .
  \]

  \pause
  \vspace{1em}
  \alert{Fairness:} ``70\%'' should mean 70\% in every group,
  e.g.\ for state-school students too.
\end{frame}
